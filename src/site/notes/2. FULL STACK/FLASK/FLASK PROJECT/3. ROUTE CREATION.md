from config import app, db

## CREATE ROUTE
```Python 
@app.route("/create", methods=['POST'])
def create_contact():
    first_Name = request.json.get("first_name")
    last_Name = request.json.get("last_name")
    email_addr = request.json.get("email")
    
    # Find whether these attr exist or not
    if not first_Name or not last_Name or not email_addr:
        return jsonify({"message": "Please add all the given details - First Name, Last Name, Email", "status_code": 400})

    # Adding new contacts
    new_contacts = Contact(first_name=first_Name,           last_name=last_Name, email=email_addr)
    try:
        db.session.add(new_contacts)
        db.session.commit()
    except Exception as e:
        return jsonify({"message": str(e), "status_code": 400})

    # Print if Everything goes fine
    return jsonify({"message": "User Created!", "status_code": 201})
```

## GET ROUTE 
```Python
# Fetch Contacts
@app.route('/contacts', methods=['GET'])
def get_contacts():
    # Call all the DATA objects from the DB.
    contacts = Contact.query.all()

    # Iterates over contacts & generate a new list
    json_contacts = list(map(lambda x: x.to_json(), contacts))

    # Converts the new_list into JSON Object
    return jsonify({"contacts": json_contacts})
```

## UPDATE ROUTE 
```Python
# Updating the existing data.
@app.route('/update_contact/<int:user_id>', methods=["PATCH"])
def update_contact(user_id):
    contact = db.session.get(Contact, user_id)

    if not contact:
        return jsonify({"message": "mentioned user doesn't exist", "status_code": 404})

    # Retriving Data from the db in json
    data = request.json
    """
    Here the data.get() works like this
        - If the new_data_var is updated i'll use it.
        - If data is new_data_var is not entered. it'll use the existing one "old_data.attr" & add it to the database.

        old_data_attr = data.get("new_data_var", old_data.attr )
    """
    contact.first_name = data.get("first_name", contact.first_name)
    contact.last_name = data.get("last_name", contact.last_name)
    contact.email = data.get("email", contact.email)
    
    db.session.commit()
    return jsonify({"message": "User details updated", "status_code": 200})
```

## DELETE ROUTE 
```Python
@app.route('/delete_contact/<int:user_id>', methods=["DELETE"])
def delete_contact(user_id):
    contact = db.session.get(Contact, user_id)

    if not contact:
        return jsonify({"message": "mentioned user doesn't exist", "status_code": 404})

    db.session.delete(contact)
    db.session.commit()
    return jsonify({"message": "User deleted", "status_code": 200})
```
