public static void IsAmstrng(){  
Scanner sc = new Scanner(System.in);  
System.out.print("Enter the Number: ");  
int num = sc.nextInt();  

int t;
t = num;
int amst;
int mul=0;

while(num!=0){  
	amst = num %10;  
	num /=10;  
	mul +=(amst*amst*amst);  
}  

if (t == mul){  
	System.out.println(t +" is armstrong");  
}  
else {  
	System.out.println(t + " is not an armstrong");  
}  

}
