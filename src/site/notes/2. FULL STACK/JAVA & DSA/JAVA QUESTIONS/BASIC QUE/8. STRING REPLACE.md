public class replace_char {  
public static void main(String[] args) {  
	Scanner sc = new Scanner(System.in);  
	System.out.print("Enter a String: ");  
	String s= sc.nextLine();  

	s = s.replaceAll("[^a-zA-Z0-9]","");  
	System.out.println(s);  
}   
}
