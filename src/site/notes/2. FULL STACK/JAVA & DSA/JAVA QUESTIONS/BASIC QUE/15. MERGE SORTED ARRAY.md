public static void merge_sorted_arr(ArrayList<Integer> arr, ArrayList<Integer> arr2){  
    ArrayList<Integer> al = new ArrayList<>();  
  
    int i =0, j=0;  
  
    // Run the loop of the arraylist size and add the element as sorted elements  
    while(i<arr.size() && j<arr2.size()) {  
        // Check if the arr[element] is smaller than arr2[element]  
        // If, yes add it in the new arraylist       
        if (arr.get(i) < arr2.get(j)) {  
            al.add(arr.get(i));  
            i++;  
        } else {  
            al.add(arr2.get(j));  
            j++;  
        }  
    }    // leftover elements to be added.  
    while(i < arr.size()){  
        al.add(arr.get(i));  
        i++;  
    }  
  
    while(j<arr2.size()){  
        al.add(arr2.get(j));  
        j++;  
    }  
System.out.println(al);  
}
