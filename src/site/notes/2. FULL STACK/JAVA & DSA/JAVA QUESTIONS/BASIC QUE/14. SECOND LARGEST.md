public int getSecondLargest(int[] arr) {
	int largest = arr[0];
	int s_largest = Integer.MIN_VALUE;
	
	for (int i = 1; i<arr.length; i++){
		if(arr[i]>largest){
			s_largest = largest;
			largest = arr[i];
		}
			else if(arr[i]<largest && arr[i]>s_largest){
			s_largest = arr[i];
		}
	}
	return (s_largest == Integer.MIN_VALUE) ? -1 : s_largest;
}
